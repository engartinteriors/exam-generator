    function handleLogoUpload(event) {

      const file = event.target.files[0];

      if (file) {

        const reader = new FileReader();

        reader.onload = function(e) {

          logoDataUrl = e.target.result;

          document.getElementById('paperLogo').src = logoDataUrl;

        };

        reader.readAsDataURL(file);

      }

    }


    function generateNewQuestions() {

      const subject = document.getElementById('subject').value;

      const grade = document.getElementById('grade').value;

      

      const pool = questionBank[subject][grade] || [];

      // Shuffle array to simulate dynamic paper generation

      currentQuestions = [...pool].sort(() => Math.random() - 0.5);

      renderPaper();

    }


    function renderPaper() {

      // Update Branding

      document.getElementById('paperSchool').innerText = document.getElementById('schoolName').value || 'Primary Academy';

      

      // Update Meta

      const subj = document.getElementById('subject').value;

      const gr = document.getElementById('grade').value;

      const syl = document.getElementById('syllabus').options[document.getElementById('syllabus').selectedIndex].text;

      

      document.getElementById('metaSubject').innerText = subj;

      document.getElementById('metaGrade').innerText = gr;

      document.getElementById('metaSyllabus').innerText = syl.split(' ')[0];

      document.getElementById('paperTitle').innerText = `${subj} — ${syl} Assessment`;


      // Render Questions

      const container = document.getElementById('questionContainer');

      container.innerHTML = '';


      if (currentQuestions.length === 0) generateNewQuestions();


      let sectionA = '<div class="section-title">Section A: Multiple Choice Questions</div>';

      let sectionB = '<div class="section-title">Section B: Short Answer & Reasoning</div>';


      currentQuestions.forEach((q, index) => {

        let qHtml = `<div class="question-item">

          <div class="question-head">

            <span>Q${index + 1}. ${q.q}</span>

            <span>[${q.m} Marks]</span>

          </div>`;


        if (q.type === 'mcq') {

          qHtml += `<div class="options-grid">

            ${q.opts.map((opt, i) => `<div>(${String.fromCharCode(65 + i)}) ${opt}</div>`).join('')}

          </div>`;

        } else {

          qHtml += `<div class="answer-space"></div>`;

        }

        

        qHtml += `</div>`;


        if (q.type === 'mcq') sectionA += qHtml;

        else sectionB += qHtml;

      });


      container.innerHTML = sectionA + sectionB;

    }


    // Timer Logic

    function updateTimerDisplay() {

      const mins = Math.floor(timerSeconds / 60);

      const secs = timerSeconds % 60;

      document.getElementById('timerDisplay').innerText = 

        `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;

    }


    function toggleTimer() {

      const btn = document.getElementById('timerBtn');

      if (isTimerRunning) {

        clearInterval(timerInterval);

        btn.innerText = "Start Timer";

      } else {

        timerInterval = setInterval(() => {

          if (timerSeconds > 0) {

            timerSeconds--;

            updateTimerDisplay();

          } else {

            clearInterval(timerInterval);

            alert("Exam Time is Up!");

          }

        }, 1000);

        btn.innerText = "Pause Timer";

      }

      isTimerRunning = !isTimerRunning;

    }


    function resetTimer() {

      clearInterval(timerInterval);

      isTimerRunning = false;

      timerSeconds = 1800;

      document.getElementById('timerBtn').innerText = "Start Timer";

      updateTimerDisplay();

    }
